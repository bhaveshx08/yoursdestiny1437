function getRandom(arr) {
    return arr[Math.floor(Math.random() * arr.length)];
}

function showStage(id) {

    document.querySelectorAll(".stage").forEach(el => {
        el.style.display = "none";
    });

    document.getElementById("home").style.display = "none";

    document.getElementById(id).style.display = "block";
}

function start() {

    name = document.getElementById("username").value.trim();

    gender = document.getElementById("gender").value;

    if (!name) {
        alert("Please enter your name!");
        return;
    }

    let message = gender === "Boy" ?
        getRandom(funnyBoy) : getRandom(funnyGirl);

    document.getElementById("funnyText").innerText =
        message.replaceAll("{name}", name);

    document.getElementById("emotionText").innerText =
        getRandom(emotions);

    showStage("funny");
}

function nextStage(stage) {

    if (stage === 2) {
        showStage("emotion");
    }

    if (stage === 3) {
        showStage("giftStage");
    }
}

function getPower() {

    let used = JSON.parse(localStorage.getItem("usedPowers") || "{}");

    let current = used[name] || [];

    if (current.length >= powers.length) {
        current = [];
    }

    let available = powers
        .map((power, index) => ({power, index}))
        .filter(item => !current.includes(item.index));

    let chosen = getRandom(available);

    current.push(chosen.index);

    used[name] = current;

    localStorage.setItem("usedPowers", JSON.stringify(used));

    return chosen.power;
}

/* FIXED GIFT FUNCTION */

function revealPower() {

    if (revealing) return;

    revealing = true;

    const gift = document.getElementById("giftBox");

    gift.disabled = true;

    gift.classList.remove("explode");

    void gift.offsetWidth;

    gift.classList.add("explode");

    setTimeout(() => {

        try {

            selectedPower = getPower();

            document.getElementById("powerIcon").innerText =
                selectedPower.icon;

            document.getElementById("powerTitle").innerText =
                selectedPower.name;

            document.getElementById("powerDescription").innerText =
                selectedPower.desc;

            showStage("powerStage");

            const powerStage = document.getElementById("powerStage");

            powerStage.classList.remove("blast");

            void powerStage.offsetWidth;

            powerStage.classList.add("blast");

            makeConfetti();

        } catch (error) {

            console.error(error);

            alert("Something went wrong! Please try again.");

            gift.disabled = false;

            gift.classList.remove("explode");

            revealing = false;
        }

    }, 800);
}

/* CONFETTI */

function makeConfetti() {

    const box = document.getElementById("confetti");

    box.innerHTML = "";

    const colors = [
        "#00eaff",
        "#ff65d5",
        "#ffe36e",
        "#8cff85",
        "#ffffff"
    ];

    for (let i = 0; i < 65; i++) {

        let piece = document.createElement("div");

        piece.className = "confetti-piece";

        piece.style.left = Math.random() * 100 + "%";

        piece.style.top = "-20px";

        piece.style.background = getRandom(colors);

        piece.style.animationDelay = Math.random() + "s";

        box.appendChild(piece);
    }

    setTimeout(() => {
        box.innerHTML = "";
    }, 3500);
}

/* RESTART */

function restart() {

    revealing = false;

    document.getElementById("home").style.display = "block";

    document.querySelectorAll(".stage").forEach(el => {
        el.style.display = "none";
    });

    document.getElementById("username").value = "";

    document.getElementById("confetti").innerHTML = "";

    const gift = document.getElementById("giftBox");

    gift.disabled = false;

    gift.classList.remove("explode");

    document.getElementById("powerStage").classList.remove("blast");
}
